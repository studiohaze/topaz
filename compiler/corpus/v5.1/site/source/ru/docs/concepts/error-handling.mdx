> **Примечание об именах библиотек**: Вспомогательные и member-вызовы вне минимума стандартной библиотеки — `Math.sqrt`, `.endsWith`, `.contains`, `файлСуществует`, `естьПравоЧтения`, `читатьСодержимоеФайла`, `создатьКонфигурациюПоУмолчанию`, `проверитьДанные`, `преобразоватьДанные`, `сохранитьДанные`, `обработатьФайл`, `пакетнаяОбработка`, `сетевойЗапрос`, `Thread.sleep`, `создатьПользователя`, `сохранитьВБазуДанных`, `запросКБазеДанных`, `увеличитьМетрикуОшибок`, `собратьЛогиОшибок`, `отправитьОповещение`, `отправитьСрочноеОповещение`, `input`, `open`, `.read`, `.close` и т.д. — являются иллюстративными placeholder, не канонические API. `JSON.parse` — зарезервированное имя стандартной библиотеки, полная типизация которого отложена.

Обработка ошибок является **основой надежного программного обеспечения**. Топаз предоставляет типы **Result** и **Option** для типобезопасной обработки ошибок. 

**Принцип «сначала Result»**  
Каноническая обработка ошибок Topaz использует `Result`, оператор распространения `?` и `match`. Формы стиля исключений не являются частью Topaz v5.2. Невосстановимые ошибки времени выполнения идут по отдельному каналу — **фолты (faults)** — он описан далее на этой странице.

##  Тип Result

### Основное использование Result

```topaz
// Result<ТипУспеха, ТипОшибки>
function разделить(числитель: float, знаменатель: float) -> Result<float, string> {
    if знаменатель == 0.0 {
        return Err("Нельзя делить на ноль")
    }
    return Ok(числитель / знаменатель)
}

// Обработка Result
let результат = разделить(10.0, 2.0)
match результат {
    case Ok(значение) => print("Результат: {значение}")
    case Err(сообщениеОшибки) => print("Ошибка: {сообщениеОшибки}")
}

// Различные случаи ошибок
function читатьФайл(путь: string) -> Result<string, string> {
    if путь == "" {
        return Err("Путь к файлу пуст")
    }
    
    if !путь.endsWith(".txt") {
        return Err("Поддерживаются только текстовые файлы")
    }
    
    if !файлСуществует(путь) {
        return Err("Файл не найден: {путь}")
    }
    
    // Фактическая логика чтения файла
    return Ok("Содержимое файла")
}
```

### Обработка Result через match

```topaz
// Преобразование успешного значения с помощью match
let результат = match разделить(10.0, 2.0) {
    case Ok(значение) => Ok("Результат: {значение * 2.0}")
    case Err(ошибка) => Err(ошибка)
}

match результат {
    case Ok(сообщение) => print(сообщение)     // сообщение об успехе
    case Err(ошибка) => print("Ошибка: {ошибка}")
}

// Цепочка обработки через вложенный match
function квадратныйКорень(значение: float) -> Result<float, string> {
    if значение < 0.0 {
        return Err("Нельзя вычислить квадратный корень отрицательного числа")
    }
    return Ok(Math.sqrt(значение))
}

let финальныйРезультат = match разделить(100.0, 4.0) {
    case Ok(значение) => match квадратныйКорень(значение) {
        case Ok(корень) => Ok("Квадратный корень: {корень}")
        case Err(ошибка) => Err(ошибка)
    }
    case Err(ошибка) => Err(ошибка)
}

match финальныйРезультат {
    case Ok(сообщение) => print(сообщение)         // сообщение о квадратном корне
    case Err(ошибка) => print("Ошибка: {ошибка}")
}

// Восстановление после ошибки через match
function использоватьПоУмолчанию(ошибка: string) -> Result<float, string> {
    print("Использование значения по умолчанию: {ошибка}")
    return Ok(0.0)
}

let восстановленныйРезультат = match разделить(10.0, 0.0) {
    case Ok(значение) => Ok(значение)
    case Err(ошибка) => использоватьПоУмолчанию(ошибка)
}

match восстановленныйРезультат {
    case Ok(значение) => print("Восстановлено: {значение}")
    case Err(ошибка) => print("Ошибка: {ошибка}")
}
```

### Сложные типы ошибок

```topaz
// Структурированные типы ошибок
type ВидОшибкиФайла = "not_found" | "permission_denied" | "read_failed" | "format_error"
type ОшибкаФайла = { вид: ВидОшибкиФайла, сообщение: string }

function читатьФайлКонфигурации(путь: string) -> Result<{ theme: string }, ОшибкаФайла> {
    if !файлСуществует(путь) {
        return Err({ вид: "not_found", сообщение: "Файл конфигурации не найден: {путь}" })
    }
    
    if !естьПравоЧтения(путь) {
        return Err({ вид: "permission_denied", сообщение: "Нет разрешения на чтение: {путь}" })
    }
    
    return match читатьСодержимоеФайла(путь) {
        case Ok(содержимое) => match JSON.parse(содержимое) {
            case Ok(конфигурация) => Ok({ theme: конфигурация.theme })
            case Err(_) => Err({ вид: "format_error", сообщение: "Неверный формат JSON" })
        }
        case Err(причина) => Err({ вид: "read_failed", сообщение: причина })
    }
}

// Специфическая обработка ошибок: сопоставляем закрытый union видов
// ошибок напрямую, чтобы исчерпываемость проверялась статически
match читатьФайлКонфигурации("config.json") {
    case Ok(конфигурация) => {
        print("Конфигурация загружена успешно")
        print("Тема: {конфигурация.theme}")
    }
    case Err(ошибка) => match ошибка.вид {
        case "not_found" => {
            print("Файл конфигурации не найден: {ошибка.сообщение}")
            создатьКонфигурациюПоУмолчанию()
        }
        case "permission_denied" => print("Нет разрешения: {ошибка.сообщение}")
        case "format_error" => print("Ошибка формата: {ошибка.сообщение}")
        case "read_failed" => print("Не удалось прочитать: {ошибка.сообщение}")
    }
}
```

##  Тип Option

### Основное использование Option

```topaz
// Option<T> - значение может существовать или не существовать
function найтиПользователя(id: int) -> Option<{ имя: string, возраст: int }> {
    let пользователи = [
        { id: 1, имя: "ТопазДев", возраст: 25 },
        { id: 2, имя: "КодМастер", возраст: 30 }
    ]
    
    for пользователь in пользователи {
        if пользователь.id == id {
            return Some({ имя: пользователь.имя, возраст: пользователь.возраст })
        }
    }
    
    return None
}

// Обработка Option
match найтиПользователя(1) {
    case Some(пользователь) => print("Пользователь найден: {пользователь.имя}, {пользователь.возраст} лет")
    case None => print("Пользователь не найден")
}

// Поиск элемента в массиве — .get является чтением без фолта
function первыйЦелочисленныйЭлемент(массив: Array<int>) -> Option<int> {
    массив.get(0)
}

let числа = [1, 2, 3, 4, 5]
match первыйЦелочисленныйЭлемент(числа) {
    case Some(первый) => print("Первый элемент: {первый}")
    case None => print("Массив пуст")
}
```

### Обработка Option через match

```topaz
// Преобразование значения с помощью match
let результат = match найтиПользователя(1) {
    case Some(пользователь) => Some("Привет, {пользователь.имя}!")
    case None => None
}

match результат {
    case Some(приветствие) => print(приветствие)
    case None => print("Пользователь не найден")
}

// Цепочка Option через вложенный match
function проверитьВзрослость(пользователь: { имя: string, возраст: int }) -> Option<string> {
    if пользователь.возраст >= 18 {
        return Some(пользователь.имя)
    }
    return None
}

let взрослыйПользователь = match найтиПользователя(1) {
    case Some(пользователь) => match проверитьВзрослость(пользователь) {
        case Some(имя) => Some("{имя} совершеннолетний")
        case None => None
    }
    case None => None
}

// Предоставление значения по умолчанию с match
let имяПользователя = match найтиПользователя(999) {
    case Some(пользователь) => пользователь.имя
    case None => "Неизвестный пользователь"
}

print("Имя пользователя: {имяПользователя}")  // "Имя пользователя: Неизвестный пользователь"

// Фильтрация по условию через match
let молодойПользователь = match найтиПользователя(1) {
    case Some(пользователь) => if пользователь.возраст < 30 { Some("{пользователь.имя} молод") } else { None }
    case None => None
}
```

### Комбинирование Option и Result

```topaz
// Преобразование Option в Result
function требуемыйПользователь(id: int) -> Result<{ имя: string, возраст: int }, string> {
    return match найтиПользователя(id) {
        case Some(пользователь) => Ok(пользователь)
        case None => Err("Пользователь не найден: ID {id}")
    }
}

// Преобразование Result в Option
let опциональныйРезультат = match разделить(10.0, 0.0) {
    case Ok(значение) => Some(значение)
    case Err(_) => None
}

match опциональныйРезультат {
    case Some(значение) => print("Деление успешно: {значение}")
    case None => print("Деление не удалось (сообщение об ошибке игнорируется)")
}

// Сложная обработка
function удвоитьВозрастПользователя(id: int) -> Result<int, string> {
    return match требуемыйПользователь(id) {
        case Ok(пользователь) => Ok(пользователь.возраст * 2)
        case Err(ошибка) => Err(ошибка)
    }
}

match удвоитьВозрастПользователя(1) {
    case Ok(удвоенныйВозраст) => print("Удвоенный возраст: {удвоенныйВозраст}")
    case Err(ошибка) => print("Ошибка: {ошибка}")
}
```

##  Распространение ошибок и досрочный возврат

### Оператор ? (распространение ошибок)

```topaz
// Автоматическое распространение ошибок с оператором ?
function сложноеВычисление(a: float, b: float, c: float) -> Result<float, string> {
    let первое = разделить(a, b)?          // Автоматический возврат при ошибке
    let второе = квадратныйКорень(первое)?  // Автоматический возврат при ошибке
    let финальное = разделить(второе, c)?   // Автоматический возврат при ошибке
    
    return Ok(финальное)
}

// Код выше эквивалентен:
function сложноеВычисление_ручное(a: float, b: float, c: float) -> Result<float, string> {
    return match разделить(a, b) {
        case Ok(первое) => match квадратныйКорень(первое) {
            case Ok(второе) => match разделить(второе, c) {
                case Ok(финальное) => Ok(финальное)
                case Err(ошибка) => Err(ошибка)
            }
            case Err(ошибка) => Err(ошибка)
        }
        case Err(ошибка) => Err(ошибка)
    }
}

// Значения Option используют явный match (`?` — только для Result)
function суммаПервогоИВторого(массив: Array<int>) -> Option<int> {
    match первыйЦелочисленныйЭлемент(массив) {
        case Some(первый) => match массив.get(1) {
            case Some(второй) => Some(первый + второй)
            case None => None
        }
        case None => None
    }
}
```

> Каноническая обработка ошибок Topaz использует распространение `?` и `match`. `?` требует значение типа `Result`; значения `Option` обрабатываются явным `match`. Блоки `try` (`try { ... }`) и выражения `try` не специфицированы в Topaz v5.2 — постфиксный `expr?` является каноническим написанием распространения.

### `defer` (охранники области видимости)

Регистрируйте действия очистки для выполнения при выходе из текущей области. Порядок LIFO. `defer` выполняется на путях выхода из области: обычное завершение, `return` и ранний возврат через `?`.

Политика: ошибки внутри действий `defer` логируются или собираются согласно политике рантайма — они никогда не заменяют уже возвращаемый `Err`, и оставшиеся отложенные действия продолжают выполняться.

```topaz
function обработатьФайл(путь: string) -> Result<string, string> {
    let файл = open(путь)?
    defer { файл.close() }                  // всегда закрывает (LIFO)

    let содержимое = файл.read()?           // работа с ресурсом
    Ok(содержимое)
}
```

##  Фолты: невосстановимый канал

Некоторые ошибки времени выполнения являются **фолтами**, а не значениями. Фолт прерывает текущее выполнение программы: это не `Result`, не `Option`, и его нельзя перехватить через `?`, `??` или `concurrent`. В Topaz v5.2 нет ключевого слова panic и нет формы перехвата фолтов.

Источники фолтов включают:

- прямую индексацию массива за пределами границ или с отрицательным индексом (`arr[i]`)
- целочисленное `/` или `%` на ноль
- переполнение целого числа в арифметике времени выполнения
- динамический шаг диапазона, равный нулю
- промах `match` во время выполнения, когда исчерпываемость не была статически доказуема и нет catch-all case

```topaz
let числа = [1, 2, 3]

let безопасно = числа.get(10)        // None — чтение без фолта
// числа[10] дало бы фолт: прямая индексация прерывает выполнение вне границ

// Храните восстановимые сбои в Result вместо форм, склонных к фолтам
function безопасныйИндекс(xs: Array<int>, i: int) -> Result<int, string> {
    match xs.get(i) {
        case Some(значение) => Ok(значение)
        case None => Err("Индекс {i} вне границ")
    }
}
```

Правило границы: используйте `Result` для сбоев, которые вызывающий код должен обработать (отсутствующие файлы, неверный ввод, сетевые ошибки); фолты зарезервированы для программных ошибок, которые никогда не должны происходить в корректном коде. Канонический Topaz моделирует восстановимые сбои через `Result` и избегает склонных к фолтам форм в публичных примерах.

##  Паттерны обработки ошибок

### Добавление контекста ошибки

```topaz
// Добавление контекстной информации к ошибкам
function загрузитьДанныеПользователя(id: int) -> Result<{ имя: string, настройки: { theme: string } }, string> {
    return match требуемыйПользователь(id) {
        case Ok(пользователь) => match читатьФайлКонфигурации("user_{id}.json") {
            case Ok(настройки) => Ok({ имя: пользователь.имя, настройки: настройки })
            case Err(ошибка) => Err("Загрузка настроек пользователя не удалась (ID: {id}): {ошибка.сообщение}")
        }
        case Err(ошибка) => Err("Поиск пользователя не удался (ID: {id}): {ошибка}")
    }
}

// Пошаговая обработка ошибок
function пошаговаяОбработка() -> Result<string, string> {
    print("Шаг 1: Проверка данных...")
    return match проверитьДанные() {
        case Err(ошибка) => Err("Шаг 1 не удался: {ошибка}")
        case Ok(результатПроверки) => {
            print("Шаг 2: Преобразование данных...")
            match преобразоватьДанные(результатПроверки) {
                case Err(ошибка) => Err("Шаг 2 не удался: {ошибка}")
                case Ok(результатПреобразования) => {
                    print("Шаг 3: Сохранение данных...")
                    match сохранитьДанные(результатПреобразования) {
                        case Err(ошибка) => Err("Шаг 3 не удался: {ошибка}")
                        case Ok(результатСохранения) => Ok("Все шаги завершены: {результатСохранения}")
                    }
                }
            }
        }
    }
}
```

### Обработка частичных сбоев

```topaz
// Разрешение частичных сбоев в нескольких операциях
function пакетнаяОбработка(списокФайлов: Array<string>) -> { успешные: Array<string>, неудачные: Array<{ файл: string, ошибка: string }> } {
    let mut списокУспешных: Array<string> = []
    let mut списокНеудачных: Array<{ файл: string, ошибка: string }> = []
    
    for файл in списокФайлов {
        match обработатьФайл(файл) {
            case Ok(результат) => {
                списокУспешных.push(результат)
                print("Успех: {файл}")
            }
            case Err(ошибка) => {
                списокНеудачных.push({ файл: файл, ошибка: ошибка })
                print("Неудача: {файл} - {ошибка}")
            }
        }
    }
    
    return { успешные: списокУспешных, неудачные: списокНеудачных }
}

// Обработка массивов Result
function собратьРезультатыFloat(результаты: Array<Result<float, string>>) -> Result<Array<float>, Array<string>> {
    let mut успешныеЗначения: Array<float> = []
    let mut ошибки: Array<string> = []
    
    for результат in результаты {
        match результат {
            case Ok(значение) => успешныеЗначения.push(значение)
            case Err(ошибка) => ошибки.push(ошибка)
        }
    }
    
    if ошибки.length > 0 {
        Err(ошибки)
    } else {
        Ok(успешныеЗначения)
    }
}

let результатыВычислений = [
    разделить(10.0, 2.0),   // Ok(5.0)
    разделить(8.0, 4.0),    // Ok(2.0)
    разделить(6.0, 0.0)     // Err("Нельзя делить на ноль")
]

match собратьРезультатыFloat(результатыВычислений) {
    case Ok(значения) => print("Все вычисления успешны: {значения}")
    case Err(ошибки) => print("Некоторые вычисления не удались: {ошибки}")
}
```

### Паттерн повторных попыток

```topaz
// Логика повторных попыток
function повторитьСтроковыйРезультат(
    операция: () -> Result<string, string>,
    максПопыток: int,
    попытка: int = 1,
    задержка: int = 1000
) -> Result<string, string> {
    return match операция() {
        case Ok(результат) => {
            if попытка > 1 {
                print("Успех на попытке {попытка}")
            }
            Ok(результат)
        }
        case Err(ошибка) => {
            print("Попытка {попытка} не удалась: {ошибка}")
            if попытка >= максПопыток {
                Err(ошибка)
            } else {
                print("Повтор через {задержка}мс...")
                Thread.sleep(задержка)
                повторитьСтроковыйРезультат(операция, максПопыток, попытка + 1, задержка)
            }
        }
    }
}

// Повторная попытка сетевого запроса
let ответ = повторитьСтроковыйРезультат(
    операция: () => сетевойЗапрос("https://api.example.com/data"),
    максПопыток: 3,
    задержка: 2000
)

match ответ {
    case Ok(данные) => print("Данные получены: {данные}")
    case Err(ошибка) => print("Окончательная неудача: {ошибка}")
```

##  Оптимизация и лучшие практики

### 1. Обращение с ошибками как со значениями

```topaz
// Хорошо: Явная обработка ошибок
function безопасныйВводПользователя() -> Result<int, string> {
    let пользовательскийВвод = input("Введите число: ")

    // toInt возвращает Option<int> — обрабатывайте случай None явно
    let число = match toInt(пользовательскийВвод) {
        case Some(значение) => значение
        case None => { return Err("Не является допустимым числом: {пользовательскийВвод}") }
    }

    if число < 0 {
        return Err("Отрицательные числа не допускаются")
    }
    return Ok(число)
}
```

Избегайте извлечения в стиле panic; храните ошибки в `Result` и обрабатывайте через `match` или `?`.

### 2. Оптимизация распространения ошибок

Используйте распространение `?` между шагами для канонической обработки `Result`, как показано в `регистрироватьПользователя` ниже.

```topaz
// Предотвращение глубокой вложенности с помощью досрочных возвратов
function регистрироватьПользователя(данные: { имя: string, email: string, возраст: int }) -> Result<string, string> {
    // Ранняя проверка
    if данные.имя == "" {
        return Err("Требуется имя")
    }
    
    if !данные.email.contains("@") {
        return Err("Требуется действительный email")
    }
    
    if данные.возраст < 0 {
        return Err("Требуется действительный возраст")
    }
    
    // Обработка только после прохождения всех проверок
    let пользователь = создатьПользователя(данные)?
    let id = сохранитьВБазуДанных(пользователь)?
    
    return Ok("Регистрация пользователя завершена: {id}")
}
```

### 3. Логирование и мониторинг ошибок

```topaz
// Структурированное логирование ошибок
function обработатьСтрокуСЛогированием(
    результат: Result<string, string>,
    контекст: string,
    уровеньЛога: string = "error"
) -> Result<string, string> {
    match результат {
        case Ok(значение) => {
            if уровеньЛога == "debug" {
                print("[DEBUG] {контекст}: Успех")
            }
            Ok(значение)
        }
        case Err(ошибка) => {
            print("[{уровеньЛога}] {контекст}: {ошибка}")
            
            // Собрать метрики
            увеличитьМетрикуОшибок(контекст, уровеньЛога)
            
            Err(ошибка)
        }
    }
}

// Пример использования
let результат = запросКБазеДанных("SELECT * FROM users")
    |> обработатьСтрокуСЛогированием(_, контекст: "Запрос списка пользователей", уровеньЛога: "error")

// Агрегация ошибок и оповещения
function системаОповещенийОбОшибках() {
    let недавниеОшибки = собратьЛогиОшибок(последниеЧасы: 1)  // 1 час
    
    if недавниеОшибки.length > 100 {
        отправитьОповещение("Обнаружен высокий уровень ошибок: {недавниеОшибки.length} ошибок/час")
    }
    
    let критическиеОшибки = filter(недавниеОшибки, ошибка => ошибка.уровень == "critical")
    if критическиеОшибки.length > 0 {
        отправитьСрочноеОповещение("Произошли критические ошибки", критическиеОшибки)
    }
}
```

Обработка ошибок в Топазе **типобезопасна и явна**. Используйте Result и Option для создания **предсказуемых и надежных** приложений!
