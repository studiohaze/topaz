> **Примечание о библиотечных именах**: вспомогательные функции и вызовы методов вне минимума стандартной библиотеки — `Math.random`, `.endsWith`, `.contains`, `.all`, `получитьПользователяИзБазы`, `валидироватьПользователя`, `обработатьДанныеПользователя`, `создатьНовогоПользователя` и т.д. — являются иллюстративными заполнителями, а не каноническими API. `JSON.parse` — зарезервированное имя стандартной библиотеки, полная типизация которого отложена.

Поток управления определяет **порядок выполнения вашей программы**. Топаз предоставляет малый замкнутый набор структур потока управления — одна форма для каждого замысла.

##  Условные операторы

### Операторы if

```topaz
let возраст = 25
let оценка = 85

// Базовый оператор if
if возраст >= 18 {
    print("Вы совершеннолетний")
}

// Оператор if-else
if оценка >= 90 {
    print("Оценка A")
} else {
    print("Нужно больше стараться")
}

// Цепочка if-else if
if оценка >= 90 {
    print("Оценка A - Отлично")
} else if оценка >= 80 {
    print("Оценка B - Хорошо")
} else if оценка >= 70 {
    print("Оценка C - Удовлетворительно")
} else {
    print("Требуется пересдача")
}

// Составные условия
let имяПользователя = "ТопазДев"
let попытки = 3

if имяПользователя != "" && попытки < 5 {
    print("Вы можете попытаться войти в систему")
}
```

### Условные выражения

```topaz
// Вместо тернарного оператора используйте if как выражение
let статусВзрослости = if возраст >= 18 { "взрослый" } else { "несовершеннолетний" }
let статусСдачи = if оценка >= 60 { "сдал" } else { "не сдал" }

// Вложенные условия
let буквеннаяОценка = if оценка >= 90 { "A" }
    else if оценка >= 80 { "B" }
    else if оценка >= 70 { "C" }
    else { "F" }

// Условное значение
let скидка = if уровеньЧленства == "VIP" { 0.2 }
    else if уровеньЧленства == "Золото" { 0.1 }
    else { 0.05 }

print("Возраст: {возраст}, Статус: {статусВзрослости}")
print("Оценка: {оценка}, Буквенная оценка: {буквеннаяОценка}")
```

##  Циклы

### Циклы for

```topaz
// Цикл for с диапазоном
for i in 0..5 {
    print("Итерация {i}")
}

// Включающий диапазон
for i in 1..10 {
    print("Число: {i}")
}

// Итерация по массиву
let фрукты = ["яблоко", "банан", "апельсин", "виноград"]
for фрукт in фрукты {
    print("Фрукт: {фрукт}")
}

// Итерация с шагом
for i in 0..20 by 2 {
    print("Четное: {i}")
}
```

### Циклы while

Topaz предоставляет общие циклы как операторы (statements). `while` повторяет своё тело, пока выполняется условие типа `bool`; условие проверяется перед каждой итерацией.

```topaz
// Накапливаем, пока не достигнем бюджета
let mut сумма = 0
let mut n = 1

while сумма + n <= 100 {
    сумма = сумма + n
    n = n + 1
}

print("Сумма в пределах бюджета: {сумма}")

// Обратный отсчёт
let mut отсчёт = 3
while отсчёт > 0 {
    print("Т-минус {отсчёт}")
    отсчёт = отсчёт - 1
}
```

`while` — это оператор, он не производит значения. Если цикл должен вернуть результат, используйте `for`-выражение, производящее значение, или рекурсию.

### Управление циклами

`break` выходит из самого внутреннего объемлющего цикла, а `continue` переходит к его следующей итерации. Оба работают внутри операторов `for` и `while`.

```topaz
let mut бюджет = 100

for стоимость in [30, -1, 25, 80, 10] {
    if стоимость < 0 { continue }       // пропускаем некорректные записи
    if стоимость > бюджет { break }     // останавливаемся на первой недоступной
    бюджет = бюджет - стоимость
    print("Куплено за {стоимость}, остаток бюджета {бюджет}")
}
```

Когда обрабатывается каждый элемент, простые исчерпывающие ветви часто яснее, чем управление циклом:

```topaz
for i in 1..20 {
    if i % 15 == 0 {
        print("{i}: FizzBuzz")
    } else if i % 3 == 0 {
        print("{i}: Fizz")
    } else if i % 5 == 0 {
        print("{i}: Buzz")
    } else {
        print("{i}")
    }
}
```

> Меченые циклы не специфицированы в Topaz v5.2. `for`, используемый как выражение, собирающее значения, не может содержать `break` или `continue`, нацеленные на него — в этом случае используйте `for`-оператор.

##  Сопоставление образцов (match)

### Базовый оператор match

```topaz
let деньНедели = "понедельник"

match деньНедели {
    case "понедельник" => print("Начало недели!")
    case "пятница" => print("Слава пятнице!")
    case день if день in ["суббота", "воскресенье"] => print("Выходные!")
    case _ => print("Обычный день")
}

// match, возвращающий значение
let выходные = ["суббота", "воскресенье"]
let типДня = match деньНедели {
    case день if день in выходные => "выходной"
    case _ => "рабочий день"
}

print("Сегодня {типДня}")
```

### Сужение типа через match (TypePattern)

```topaz
let rawInput = "42"
let значение = JSON.parse(rawInput)

match значение {
    case текст: string => print("Текст: {текст}")
    case число: int => print("Число: {число}")
    case флаг: bool => print("Флаг: {флаг}")
    case _ => print("Неподдерживаемое значение")
}
```

### Сопоставление чисел

```topaz
let оценка = 87

let буквеннаяОценка = match оценка {
    case 90..100 => "A"
    case 80..<90 => "B"
    case 70..<80 => "C"
    case 60..<70 => "D"
    case _ => "F"
}

// Использование условий-ограждений
let оценивание = match оценка {
    case x if x >= 95 => "Идеальный результат!"
    case x if x >= 90 => "Отличная работа"
    case x if x >= 80 => "Хороший результат"
    case x if x >= 70 => "Средний уровень"
    case x if x >= 60 => "Сдано"
    case _ => "Требуется пересдача"
}

print("Оценка: {оценка}, Буквенная оценка: {буквеннаяОценка}, Оценка: {оценивание}")
```

### Сопоставление записей

```topaz
let пользователь = {
    имя: "ТопазДев",
    возраст: 25,
    работа: "разработчик",
    местоположение: "Москва"
}

// Деструктурирующее сопоставление записи
match пользователь {
    case { имя: "ТопазДев", возраст, работа: "разработчик" } => {
        print("Разработчик на Топазе, возраст: {возраст}")
    }
    case { возраст: пользовательскийВозраст, местоположение: "Москва" } if пользовательскийВозраст < 30 => {
        print("Молодой пользователь в Москве")
    }
    case { работа: "разработчик", местоположение } => {
        print("Разработчик в {местоположение}")
    }
    case _ => {
        print("Обычный пользователь")
    }
}

// Сопоставление вложенных записей
let заказ = {
    клиент: { имя: "ПользовательТопаз", vip: true },
    товары: ["ноутбук", "мышь"],
    общаяСумма: 1500000
}

match заказ {
    case { клиент: { vip: true }, общаяСумма: сумма } if сумма > 1000000 => {
        print("Крупный заказ VIP-клиента: {сумма} рублей")
    }
    case { клиент: { имя }, товары } if товары.length > 1 => {
        print("Заказ {имя} с несколькими товарами")
    }
    case _ => {
        print("Обычный заказ")
    }
}
```

### Сопоставление массивов

```topaz
let числа = [1, 2, 3, 4, 5]

match числа {
    case [] => print("Пустой массив")
    case [первый] => print("Один элемент: {первый}")
    case [первый, второй] => print("Два элемента: {первый}, {второй}")
    case [первый, ..остальные] => {
        print("Первый: {первый}, Остальные: {остальные}")
    }
}

// Специфичное сопоставление образцов
let списокОценок = [95, 87, 92]

match списокОценок {
    case [100, ..] => print("Начинается с идеальной оценки!")
    case оценки if оценки.length > 0 && оценки[оценки.length - 1] == 100 => print("Заканчивается идеальной оценкой!")
    case [a, b, c] if a > 90 && b > 90 && c > 90 => print("Все оценки выше 90!")
    case оценки if filter(оценки, x => x >= 80).length == оценки.length => print("Все оценки выше 80")
    case _ => print("Обычное распределение оценок")
}
```

### Сопоставление Option и Result

```topaz
// Сопоставление типа Option
let опциональноеЗначение: Option<string> = Some("Топаз")

match опциональноеЗначение {
    case Some(значение) => print("Имеет значение: {значение}")
    case None => print("Нет значения")
}

// Сопоставление типа Result
function разделить(a: int, b: int) -> Result<int, string> {
    if b == 0 {
        return Err("Нельзя делить на ноль")
    }
    return Ok(a / b)
}

let числитель = 10
let знаменатель = 2

match разделить(числитель, знаменатель) {
    case Ok(результат) => print("Результат деления: {результат}")
    case Err(ошибка) => print("Ошибка: {ошибка}")
}
```

> Пользовательские sum types с конструкторами не специфицированы в Topaz v5.2. Для простых tagged state используйте union строковых литералов (`type Статус = "ожидание" | "обработка" | "завершено" | "ошибка"`), `Option<T>` для nullable-паттернов и `Result<T, E>` для error-or-success паттернов.

##  Условия-ограждения и продвинутые образцы

### Сложные условия-ограждения

```topaz
let температура = 25
let влажность = 60
let сезон = "лето"

match { температура: температура, влажность: влажность, сезон: сезон } {
    case { температура, влажность, сезон: "лето" } if температура > 30 && влажность > 70 => print("Очень жарко и влажно")
    case { температура, влажность, сезон: "лето" } if температура > 25 && влажность < 50 => print("Тепло и сухо")
    case { температура, сезон: "зима" } if температура < 0 => print("Очень холодно")
    case { температура, влажность } if температура >= 20 && температура <= 25 && влажность >= 40 && влажность <= 60 => print("Приятная погода")
    case _ => print("Обычная погода")
}
```

### Функциональное сопоставление образцов

```topaz
// Рекурсивное сопоставление образцов для обработки списка
function суммаСписка(список: Array<int>) -> int {
    match список {
        case [] => 0
        case [первый, ..остальные] => первый + суммаСписка(остальные)
    }
}

print("Сумма списка: {суммаСписка([1, 2, 3, 4, 5])}")  // 15
```

> Рекурсия по `Array<T>` канонична, и Topaz поддерживает дженерик-объявления функций ранга 1 (в форме `function суммаСписка<T>(...)`, параметры типов выводятся в точках вызова — см. страницу типов данных). Пользовательские generic sum types (например, варианты бинарного дерева) не специфицированы в Topaz v5.2.

##  Продвинутые образцы потока управления

### Комбинирование циклов с сопоставлением образцов

```topaz
let списокЗадач = [
    { тип: "email", приоритет: 1, содержание: "Проверить расписание встреч" },
    { тип: "звонок", приоритет: 3, содержание: "Консультация клиента" },
    { тип: "документ", приоритет: 2, содержание: "Написать отчет" }
]

for задача in списокЗадач {
    match задача {
        case { тип: "email", приоритет: 1, содержание } => {
            print("Срочная обработка email: {содержание}")
        }
        case { тип: типЗадачи, приоритет, содержание } if приоритет <= 2 => {
            print("Важная задача {типЗадачи}: {содержание}")
        }
        case { тип, содержание } => {
            print("Обычная задача {тип}: {содержание}")
        }
    }
}
```

### Обработка ошибок и поток управления

```topaz
// Обработка ошибок в стиле Result-first
function безопасноеЧтениеФайла(имяФайла: string) -> Result<string, string> {
    if !имяФайла.endsWith(".txt") {
        return Err("Поддерживаются только текстовые файлы")
    }
    
    // Симуляция чтения файла
    if Math.random() > 0.7 {
        return Err("Файл не найден")
    }
    
    return Ok("Содержимое файла здесь")
}

// Поток управления с обработкой ошибок
for имяФайла in ["data.txt", "config.json", "readme.txt"] {
    match безопасноеЧтениеФайла(имяФайла) {
        case Ok(содержимое) => {
            print("Успешно прочитан: {имяФайла}")
            print("Содержимое: {содержимое}")
            // Путь успеха для этого файла
        }
        case Err(сообщениеОшибки) => {
            print("Ошибка ({имяФайла}): {сообщениеОшибки}")
            if сообщениеОшибки.contains("поддерживаются") {
                print("Неподдерживаемый формат файла, отмечено")
            } else {
                print("Критическая ошибка отмечена")
            }
        }
    }
}
```

### Цепочки условного выполнения

```topaz
// Условное связывание
function обработатьПользователя(идПользователя: int) -> Result<string, string> {
    let данныеПользователя = match получитьПользователяИзБазы(идПользователя) {
        case Ok(данные) => данные
        case Err(ошибка) => { return Err("Не удалось получить пользователя: {ошибка}") }
    }
    
    let результатВалидации = match валидироватьПользователя(данныеПользователя) {
        case Ok(_) => ()
        case Err(ошибка) => { return Err("Валидация не пройдена: {ошибка}") }
    }
    
    let результатОбработки = match обработатьДанныеПользователя(данныеПользователя) {
        case Ok(результат) => результат
        case Err(ошибка) => { return Err("Обработка не удалась: {ошибка}") }
    }
    
    return Ok("Обработка завершена: {результатОбработки}")
}

```

##  Оптимизация и лучшие практики

### 1. Оптимизация условных операторов

```topaz
// Плохо: Вложенные условия
let оценка = 85
if оценка >= 90 {
    print("Оценка A")
} else {
    if оценка >= 80 {
        print("Оценка B")
    } else {
        if оценка >= 70 {
            print("Оценка C")
        } else {
            print("Оценка F")
        }
    }
}

// Хорошо: Использование match
let буквеннаяОценка = match оценка {
    case 90..100 => "A"
    case 80..<90 => "B"
    case 70..<80 => "C"
    case _ => "F"
}
print("Оценка {буквеннаяОценка}")
```

### 2. Образец раннего возврата

```topaz
function зарегистрироватьПользователя(имя: string, email: string, возраст: int) -> Result<string, string> {
    // Ранняя валидация и возврат
    if имя == "" {
        return Err("Требуется имя")
    }
    
    if !email.contains("@") {
        return Err("Требуется действительный email")
    }
    
    if возраст < 0 || возраст > 120 {
        return Err("Требуется действительный возраст")
    }
    
    // Обработка только при прохождении всех валидаций
    let пользователь = создатьНовогоПользователя(имя, email, возраст)
    return Ok("Регистрация пользователя завершена: {пользователь.id}")
}
```

### 3. Использование сопоставления образцов

```topaz
// Выражение сложной бизнес-логики с помощью образцов
type СтатусЗаказа = "создан" | "оплата_завершена" | "доставка" | "доставлен" | "отменен"

function обработатьЗаказ(статус: СтатусЗаказа, суммаОплаты: int, этоVip: bool) -> string {
    match { статус: статус, сумма: суммаОплаты, vip: этоVip } {
        case { статус: "создан" } => "Пожалуйста, проведите оплату"
        case { статус: "оплата_завершена", сумма, vip: true } if сумма > 100000 => {
            "Подготовка экспресс-доставки для VIP-клиента"
        }
        case { статус: "оплата_завершена", сумма } if сумма > 50000 => {
            "Подготовка обычной доставки"
        }
        case { статус: "оплата_завершена" } => "Подготовка стандартной доставки"
        case { статус: "доставка", vip: true } => "Служба отслеживания VIP-доставки"
        case { статус: "доставка" } => "В доставке"
        case { статус: "доставлен" } => "Доставка завершена"
        case { статус: "отменен" } => "Заказ отменен"
        case _ => "Неизвестное состояние заказа"
    }
}
```

> Когда scrutinee `match` — это построенная на месте запись, как здесь, исчерпываемость может быть статически недоказуемой — промах во время выполнения является фолтом, поэтому канонические примеры сохраняют catch-all case. Сопоставление непосредственно со значением закрытого литерального union (например, с самим `СтатусЗаказа`) проверяется на исчерпываемость статически и не требует catch-all.

```topaz
function ярлыкДоставки(статус: СтатусЗаказа) -> string {
    match статус {
        case "создан" => "Ожидает оплаты"
        case "оплата_завершена" => "Готовится к отправке"
        case "доставка" => "В пути"
        case "доставлен" => "Доставлен"
        case "отменен" => "Отменён"
    }
}
```

Поток управления Топаза **мал и предсказуем**. Правильно комбинируйте условные операторы, циклы и сопоставление образцов, чтобы писать **читаемый и поддерживаемый** код!
