> **Примечание об именах библиотек**: Вспомогательные и member-вызовы вне минимума стандартной библиотеки — `Math.random`, `.split`, `.trim`, `.contains` и т.д. — являются иллюстративными placeholder, не канонические API.

Функции являются **гражданами первого класса** в Топазе. Вы можете сохранять их в переменных, передавать в качестве параметров другим функциям и возвращать из функций.

В позициях типов публичная документация Topaz использует запись `(T) -> U`. Блоки могут неявно возвращать последнее выражение, но в этом руководстве для многошаговых примеров предпочтён явный `return`.

##  Базовое объявление функций

### Стандартное объявление функции

```topaz
// Базовая структура функции
function приветствовать(имя: string) -> string {
    return "Привет, {имя}!"
}

// Функция без параметров
function получитьТекущееВремя() -> string {
    return "20 декабря 2024"
}

// Функция без возвращаемого значения (unit ())
function вывестиСообщение(сообщение: string) {
    print("Лог: {сообщение}")
}

// Примеры использования
let приветствие = приветствовать("Топаз Разработчик")
print(приветствие)  // "Привет, Топаз Разработчик!"
```

### Параметры и значения по умолчанию

```topaz
// Значения параметров по умолчанию
function вычислить(a: int, b: int = 10, операция: string = "сложение") -> int {
    return match операция {
        case "сложение" => a + b
        case "вычитание" => a - b
        case "умножение" => a * b
        case _ => 0
    }
}

// Различные способы вызова
let результат1 = вычислить(5)                     // 5 + 10 = 15
let результат2 = вычислить(5, 3)                  // 5 + 3 = 8
let результат3 = вычислить(5, 3, "умножение")     // 5 * 3 = 15

// Именованные параметры
let результат4 = вычислить(a: 8, операция: "вычитание", b: 3)  // 8 - 3 = 5
```

### Вариативные параметры

```topaz
// Вариативные параметры
function сумма(...числа: int) -> int {
    let mut итого = 0
    for число in числа {
        итого += число
    }
    return итого
}

let итого1 = сумма(1, 2, 3, 4, 5)        // 15
let итого2 = сумма(10, 20)               // 30

// Комбинирование фиксированного параметра с вариативным хвостом
function вычислитьСтатистику(метка: string, ...значения: float) -> string {
    let количество = значения.length
    let mut итого = 0.0
    for значение in значения {
        итого += значение
    }
    return "{метка}: количество={количество}, итого={итого}"
}

print(вычислитьСтатистику("Результаты тестов", 85.5, 92.0, 78.5, 89.0))

// Spread массива в вариативный хвост
let оценки = [85.5, 92.0, 78.5, 89.0]
print(вычислитьСтатистику("Результаты тестов", ...оценки))

// Типы функций записывают вариативный хвост так же
let функцияСтатистики: (string, ...float) -> string = вычислитьСтатистику
```

Каноническая вариативная запись — `...args: T`, параметр собирает аргументы в `Array<T>`. Spread-аргумент `...expr` должен быть `Array` и может питать только вариативный хвост — он не может заполнять фиксированные позиционные параметры.

```topaz
// сумма собирает в Array<int>; spread заполняет хвост из массива
let части = [10, 20]
let итого3 = сумма(...части)             // 30
let итого4 = сумма(1, ...части, 4)       // 35
```

##  Стрелочные функции

### Базовые стрелочные функции

```topaz
// Простая стрелочная функция
let квадрат = (x: int) => x * x
let приветствие = (имя: string) => "Привет, {имя}!"

// Стрелочная функция без параметров
let случайноеЧисло = () => Math.random() * 100.0

// Сложная стрелочная функция
let информацияПользователя = (имя: string, возраст: int) => {
    let статус = if возраст >= 18 { "взрослый" } else { "несовершеннолетний" }
    return {
        имя: имя,
        возраст: возраст,
        статус: статус
    }
}

// Примеры использования
print("{квадрат(5)}")                   // 25
print(приветствие("ТопазДев"))          // "Привет, ТопазДев!"
let информация = информацияПользователя("Иван", 25)
print("{информация.имя} — {информация.статус}")  // "Иван — взрослый"
```

### Методы массивов со стрелочными функциями

```topaz
let числа = [1, 2, 3, 4, 5]

// map преобразование
let квадраты = map(числа, x => x * x)
print("{квадраты}")  // [1, 4, 9, 16, 25]

// filter фильтрация
let четные = filter(числа, x => x % 2 == 0)
print("{четные}")  // [2, 4]

// reduce сведение (начальное значение идёт перед редьюсером)
let итого = reduce(числа, 0, (аккум, текущий) => аккум + текущий)
print("{итого}")   // 15

// Сложное связывание
let результат = числа
    |> filter(_, x => x > 2)
    |> map(_, x => x * 3)
    |> reduce(_, 0, (a, b) => a + b)
print("{результат}")   // (3 + 4 + 5) * 3 = 36
```

##  Функции высшего порядка

### Функции как параметры

```topaz
// Определение функции высшего порядка
function применитьОперацию(массив: Array<int>, операция: (int) -> int) -> Array<int> {
    let mut результат: Array<int> = []
    for элемент in массив {
        результат.push(операция(элемент))
    }
    return результат
}

// Передача функций как параметров
let числа = [1, 2, 3, 4]
let результатКвадратов = применитьОперацию(числа, x => x * x)
let результатУдвоения = применитьОперацию(числа, x => x * 2)

print("{результатКвадратов}")  // [1, 4, 9, 16]
print("{результатУдвоения}")   // [2, 4, 6, 8]

// Функция условной обработки
function условнаяОбработка(значение: int, условие: (int) -> bool, обработчик: (int) -> int) -> int {
    if условие(значение) {
        return обработчик(значение)
    }
    return значение
}

let результат = условнаяОбработка(15, x => x > 10, x => x * 2)
print("{результат}")  // 30 (поскольку 15 > 10, то 15 * 2)
```

### Возврат функций

```topaz
// Фабрика функций
function создатьУмножитель(множитель: int) -> (int) -> int {
    return (x: int) => x * множитель
}

let удвоитель = создатьУмножитель(2)
let утроитель = создатьУмножитель(3)

print("{удвоитель(5)}")  // 10
print("{утроитель(4)}")  // 12

// Настраиваемая функция валидации
function создатьВалидаторДиапазона(мин: int, макс: int) -> (int) -> bool {
    return (значение: int) => значение >= мин && значение <= макс
}

let валидаторВозрастаВзрослых = создатьВалидаторДиапазона(18, 65)
let валидаторВозрастаСтудентов = создатьВалидаторДиапазона(5, 18)

print("{валидаторВозрастаВзрослых(25)}")  // true
print("{валидаторВозрастаСтудентов(16)}") // true
print("{валидаторВозрастаСтудентов(25)}") // false
```

### Дженерик-функции

Topaz поддерживает дженерик-объявления функций ранга 1. Параметры типов выводятся в каждой точке вызова — явные аргументы типов вроде `применитьДважды<int>(...)` не входят в Topaz v5.2; фиксируйте тип аннотированной привязкой, параметром или позицией возврата (дженерик-псевдонимы типов — см. страницу типов данных).

```topaz
function применитьДважды<T>(значение: T, преобразование: (T) -> T) -> T {
    преобразование(преобразование(значение))
}

let пять = применитьДважды(3, x => x + 1)             // T выводится как int
let громко = применитьДважды("topaz", x => "{x}!")    // T выводится как string

print("{пять}")   // 5
print(громко)     // "topaz!!"
```

Границы дженериков, ограничения и интерфейсы остаются отложенными.

##  Замыкания

### Базовые замыкания

```topaz
function создатьСчетчик(начальноеЗначение: int) -> () -> int {
    let mut счет = начальноеЗначение  // Захватывается замыканием
    
    return () => {
        счет += 1      // Доступ к внешней переменной
        счет
    }
}

let счетчик1 = создатьСчетчик(0)
let счетчик2 = создатьСчетчик(100)

print("{счетчик1()}")  // 1
print("{счетчик1()}")  // 2
print("{счетчик2()}")  // 101
print("{счетчик1()}")  // 3
```

### Сложные паттерны замыканий

```topaz
// Замыкание с состоянием
function создатьСчет(начальныйБаланс: float) -> { пополнить: (float) -> float, снять: (float) -> float, проверитьБаланс: () -> float } {
    let mut баланс = начальныйБаланс
    
    return {
        пополнить: (сумма: float) => {
            баланс += сумма
            баланс
        },
        снять: (сумма: float) => {
            if сумма <= баланс {
                баланс -= сумма
            }
            баланс
        },
        проверитьБаланс: () => баланс
    }
}

let мойСчет = создатьСчет(1000.0)
print("{мойСчет.пополнить(500.0)}")     // 1500.0
print("{мойСчет.снять(200.0)}")         // 1300.0
print("{мойСчет.проверитьБаланс()}")    // 1300.0

// Замыкание, захватывающее окружение
type Конфигурация = { тема: string, язык: string, уведомления: bool }

function создатьМенеджерКонфигурации(базоваяКонфигурация: Конфигурация) -> { установитьТему: (string) -> string, тема: () -> string } {
    let mut текущаяКонфигурация = базоваяКонфигурация
    
    return {
        установитьТему: (тема: string) => {
            текущаяКонфигурация = текущаяКонфигурация{ тема: тема }
            текущаяКонфигурация.тема
        },
        тема: () => текущаяКонфигурация.тема
    }
}

let конфигурацияПриложения = создатьМенеджерКонфигурации({
    тема: "темная",
    язык: "Русский",
    уведомления: true
})

print(конфигурацияПриложения.тема())          // "темная"
конфигурацияПриложения.установитьТему("светлая")
print(конфигурацияПриложения.тема())          // "светлая"
```

##  Рекурсивные функции

### Базовая рекурсия

```topaz
// Вычисление факториала
function факториал(n: int) -> int {
    if n <= 1 {
        return 1
    }
    return n * факториал(n - 1)
}

print("{факториал(5)}")  // 120

// Последовательность Фибоначчи
function фибоначчи(n: int) -> int {
    if n <= 1 {
        return n
    }
    return фибоначчи(n - 1) + фибоначчи(n - 2)
}

print("{фибоначчи(8)}")  // 21

// Сумма массива (рекурсия по индексу — в §22 нет срезов)
function суммаС(массив: Array<int>, начало: int) -> int {
    return match массив.get(начало) {
        case Some(значение) => значение + суммаС(массив, начало + 1)
        case None => 0
    }
}

function суммаМассива(массив: Array<int>) -> int {
    return суммаС(массив, 0)
}

print("{суммаМассива([1, 2, 3, 4, 5])}")  // 15
```

### Оптимизация хвостовой рекурсии

```topaz
// Хвостово-рекурсивный факториал
function хвостовойФакториал(n: int, аккумулятор: int = 1) -> int {
    if n <= 1 {
        return аккумулятор
    }
    return хвостовойФакториал(n - 1, аккумулятор * n)  // Хвостовой вызов
}

// Хвостово-рекурсивный Фибоначчи
function хвостовойФибоначчи(n: int, a: int = 0, b: int = 1) -> int {
    if n == 0 {
        return a
    }
    return хвостовойФибоначчи(n - 1, b, a + b)  // Хвостовой вызов
}

print("{хвостовойФакториал(10)}")  // 3628800
print("{хвостовойФибоначчи(10)}")  // 55
```

##  Паттерны функционального программирования

### Каррирование

```topaz
// Каррированная функция
function сложить(a: int) -> (int) -> int {
    return (b: int) => a + b
}

let прибавитьПять = сложить(5)
print("{прибавитьПять(3)}")  // 8

// Многопараметровое каррирование
function умножитьИСложить(множитель: float) -> (float) -> (float) -> float {
    return (слагаемое: float) => (значение: float) => значение * множитель + слагаемое
}

let удвоитьИПрибавитьДесять = умножитьИСложить(2.0)(10.0)
print("{удвоитьИПрибавитьДесять(5.0)}")  // 20.0 (5 * 2 + 10)
```

### Композиция функций

```topaz
let квадрат = (x: int) => x * x
let удвоить = (x: int) => x * 2
let меткаЦелого = (x: int) => "значение: {x}"

// Композиция через прокачку значения по этапам
let квадратЗатемУдвоить = (x: int) => x |> квадрат() |> удвоить()
let квадратЗатемУдвоитьЗатемМетка = (x: int) => x |> квадрат() |> удвоить() |> меткаЦелого()

print("{квадратЗатемУдвоить(3)}")        // 18 (3² * 2)
print(квадратЗатемУдвоитьЗатемМетка(4))  // "значение: 32" (4² * 2 → метка)
```

Публичная идиома композиции в Topaz — конвейер `|>`, как выше. Специальный оператор композиции существует только на уровне SPEC и не входит ни в одну текущую поверхность profile — см. справочник операторов.

### Мемоизация

```topaz
type МемоЗапись = { вход: int, выход: int }

function найтиВКеше(кеш: Array<МемоЗапись>, вход: int) -> Option<int> {
    match кеш {
        case [] => None
        case [первая, ..остаток] => {
            if первая.вход == вход {
                Some(первая.выход)
            } else {
                найтиВКеше(остаток, вход)
            }
        }
    }
}

// Декоратор мемоизации для функций int -> int
function мемоизироватьЦелое(функция: (int) -> int) -> (int) -> int {
    let mut кеш: Array<МемоЗапись> = []

    return (вход: int) => {
        match найтиВКеше(кеш, вход) {
            case Some(выход) => выход
            case None => {
                let результат = функция(вход)
                кеш.push({ вход: вход, выход: результат })
                результат
            }
        }
    }
}

function медленныйФибоначчи(n: int) -> int {
    if n <= 1 { n } else { медленныйФибоначчи(n - 1) + медленныйФибоначчи(n - 2) }
}

// Оптимизация медленного Фибоначчи с мемоизацией
let мемоФибоначчи = мемоизироватьЦелое(медленныйФибоначчи)

// Теперь большие числа вычисляются быстро
print("{мемоФибоначчи(20)}")  // Вычисляется быстро
```

> Topaz поддерживает дженерик-объявления функций ранга 1 (см. «Дженерик-функции» выше), но `мемоизироватьЦелое` намеренно остаётся конкретным: дженерик-`memoize<T, R>` потребовал бы сравнения на равенство для `T` при поиске в кеше, а границы и ограничения дженериков остаются отложенными. Тот же паттерн можно применять к другим конкретным типам через отдельные функции.

##  Оптимизация функций и лучшие практики

### 1. Использование чистых функций

```topaz
// Хорошо: Чистая функция
function вычислитьНалог(доход: float, налоговаяСтавка: float) -> float {
    return доход * налоговаяСтавка
}

// Избегайте: Функция с побочными эффектами
let mut глобальныйСчетчик = 0
function плохаяФункция(значение: int) -> int {
    глобальныйСчетчик += 1  // Побочный эффект!
    return значение * 2
}

// Хорошая альтернатива: Возврат состояния
function хорошаяФункция(значение: int, текущийСчетчик: int) -> { результат: int, новыйСчетчик: int } {
    return {
        результат: значение * 2,
        новыйСчетчик: текущийСчетчик + 1
    }
}
```

### 2. Именование функций и структурирование

```topaz
// Четкие имена функций
function валидироватьEmailПользователя(email: string) -> bool {
    return email.contains("@") && email.contains(".")
}

function вычислитьОбщуюСуммуЗаказа(элементыЗаказа: Array<{ ценаВКопейках: int, количество: int }>) -> int {
    return элементыЗаказа
        |> map(_, элемент => элемент.ценаВКопейках * элемент.количество)
        |> reduce(_, 0, (общее, сумма) => общее + сумма)
}

// Принцип единственной ответственности
function разобратьДанныеПользователя(исходныеДанные: string) -> { имя: string, возраст: int } {
    // Отвечает только за разбор — toInt возвращает Option<int>
    let части = исходныеДанные.split(",")
    let возраст = match toInt(части[1].trim()) {
        case Some(значение) => значение
        case None => 0
    }
    return {
        имя: части[0].trim(),
        возраст: возраст
    }
}

function валидироватьДанныеПользователя(пользователь: { имя: string, возраст: int }) -> bool {
    // Отвечает только за валидацию
    return пользователь.имя != "" && пользователь.возраст > 0
}
```

### 3. Обработка ошибок

```topaz
// Безопасная функция с типом Result
function безопасноеДеление(числитель: float, знаменатель: float) -> Result<float, string> {
    if знаменатель == 0.0 {
        return Err("Нельзя делить на ноль")
    }
    return Ok(числитель / знаменатель)
}

// Использование
match безопасноеДеление(10.0, 2.0) {
    case Ok(результат) => print("Результат: {результат}")
    case Err(ошибка) => print("Ошибка: {ошибка}")
}
```

Функции и замыкания являются **основными возможностями** Топаз. Их эффективное использование позволяет писать **переиспользуемый и тестируемый** код!
