> **라이브러리 이름 참고**: 표준 라이브러리 최소 표면을 벗어나는 도우미·멤버 호출 — `Math.random`, `.split`, `.trim`, `.contains` 등 — 은 설명용 placeholder이며 canonical API가 아닙니다.

함수는 토파즈에서 **1등 시민**입니다. 변수에 저장하고, 다른 함수의 매개변수로 전달하고, 함수에서 반환할 수 있습니다.

타입 위치에서 Topaz 공개 문서는 `(T) -> U` 표기를 사용합니다. 블록은 마지막 표현식을 암묵적으로 반환할 수 있지만, 이 가이드에서는 여러 단계가 있는 예제에서 가독성을 위해 `return`을 선호합니다.

##  기본 함수 선언

### 표준 함수 선언

```topaz
// 기본 함수 구조
function 인사하기(이름: string) -> string {
    return "안녕하세요, {이름}님!"
}

// 매개변수 없는 함수
function 현재시간() -> string {
    return "2024년 12월 20일"
}

// 반환값 없는 함수 (unit ())
function 로그출력(메시지: string) {
    print("로그: {메시지}")
}

// 사용 예시
let 인사말 = 인사하기("토파즈 개발자")
print(인사말)  // "안녕하세요, 토파즈 개발자님!"
```

### 매개변수와 기본값

```topaz
// 기본 매개변수값
function 계산하기(a: int, b: int = 10, 연산: string = "더하기") -> int {
    return match 연산 {
        case "더하기" => a + b
        case "빼기" => a - b
        case "곱하기" => a * b
        case _ => 0
    }
}

// 다양한 호출 방식
let 결과1 = 계산하기(5)           // 5 + 10 = 15
let 결과2 = 계산하기(5, 3)        // 5 + 3 = 8
let 결과3 = 계산하기(5, 3, "곱하기")  // 5 * 3 = 15

// 명명된 매개변수
let 결과4 = 계산하기(a: 8, 연산: "빼기", b: 3)  // 8 - 3 = 5
```

### 가변 매개변수

```topaz
// 가변 매개변수
function 합계구하기(...숫자들: int) -> int {
    let mut 합계 = 0
    for 숫자 in 숫자들 {
        합계 += 숫자
    }
    return 합계
}

let 총합1 = 합계구하기(1, 2, 3, 4, 5)        // 15
let 총합2 = 합계구하기(10, 20)               // 30

// 고정 매개변수와 가변 꼬리 결합
function 통계계산(레이블: string, ...값들: float) -> string {
    let 개수 = 값들.length
    let mut 합계 = 0.0
    for 값 in 값들 {
        합계 += 값
    }
    return "{레이블}: 개수={개수}, 합계={합계}"
}

print(통계계산("시험점수", 85.5, 92.0, 78.5, 89.0))

// 배열을 가변 꼬리로 spread
let 점수배열 = [85.5, 92.0, 78.5, 89.0]
print(통계계산("시험점수", ...점수배열))

// 함수 타입도 같은 방식으로 가변 꼬리를 표기합니다
let 통계함수: (string, ...float) -> string = 통계계산
```

canonical 가변 매개변수 표기는 `...args: T`이며, 매개변수는 인자들을 `Array<T>`로 수집합니다. spread 인자 `...expr`은 `Array`여야 하고 가변 꼬리에만 공급할 수 있습니다 — 고정 위치 매개변수를 채울 수 없습니다.

```topaz
// 합계구하기는 Array<int>로 수집; spread로 배열에서 다시 채울 수 있음
let 부분들 = [10, 20]
let 총합3 = 합계구하기(...부분들)             // 30
let 총합4 = 합계구하기(1, ...부분들, 4)       // 35
```

##  화살표 함수

### 기본 화살표 함수

```topaz
// 간단한 화살표 함수
let 제곱 = (x: int) => x * x
let 인사 = (이름: string) => "안녕, {이름}!"

// 매개변수 없는 화살표 함수
let 랜덤숫자 = () => Math.random() * 100.0

// 복잡한 화살표 함수
let 사용자정보 = (이름: string, 나이: int) => {
    let 상태 = if 나이 >= 18 { "성인" } else { "미성년자" }
    return {
        이름: 이름,
        나이: 나이,
        상태: 상태
    }
}

// 사용 예시
print("{제곱(5)}")               // 25
print(인사("김토파즈"))            // "안녕, 김토파즈!"
let 정보 = 사용자정보("이개발", 25)
print("{정보.이름}님은 {정보.상태}입니다")  // "이개발님은 성인입니다"
```

### 배열 메서드와 화살표 함수

```topaz
let 숫자들 = [1, 2, 3, 4, 5]

// map 변환
let 제곱들 = map(숫자들, x => x * x)
print("{제곱들}")  // [1, 4, 9, 16, 25]

// filter 필터링
let 짝수들 = filter(숫자들, x => x % 2 == 0)
print("{짝수들}")  // [2, 4]

// reduce 축약 (초기값이 reducer보다 앞에 옵니다)
let 합계 = reduce(숫자들, 0, (누적, 현재) => 누적 + 현재)
print("{합계}")   // 15

// 복잡한 체이닝
let 결과 = 숫자들
    |> filter(_, x => x > 2)
    |> map(_, x => x * 3)
    |> reduce(_, 0, (a, b) => a + b)
print("{결과}")   // (3 + 4 + 5) * 3 = 36
```

##  고차 함수

### 함수를 매개변수로 받기

```topaz
// 고차 함수 정의
function 연산적용(배열: Array<int>, 연산함수: (int) -> int) -> Array<int> {
    let mut 결과: Array<int> = []
    for 요소 in 배열 {
        결과.push(연산함수(요소))
    }
    return 결과
}

// 함수를 매개변수로 전달
let 숫자들 = [1, 2, 3, 4]
let 제곱결과 = 연산적용(숫자들, x => x * x)
let 두배결과 = 연산적용(숫자들, x => x * 2)

print("{제곱결과}")  // [1, 4, 9, 16]
print("{두배결과}")  // [2, 4, 6, 8]

// 조건부 처리 함수
function 조건부처리(값: int, 조건: (int) -> bool, 처리함수: (int) -> int) -> int {
    if 조건(값) {
        return 처리함수(값)
    }
    return 값
}

let 결과 = 조건부처리(15, x => x > 10, x => x * 2)
print("{결과}")  // 30 (15 > 10이므로 15 * 2)
```

```topaz
// 함수 팩토리
function 곱셈기만들기(배수: int) -> (int) -> int {
    return (x: int) => x * 배수
}

let 두배만들기 = 곱셈기만들기(2)

function 범위검사기만들기(최소: int, 최대: int) -> (int) -> bool {
    return (값: int) => 값 >= 최소 && 값 <= 최대

let 성인나이검사 = 범위검사기만들기(18, 65)

print("{학생나이검사(25)}")  // false
```

### 제네릭 함수

Topaz는 랭크-1 제네릭 함수 선언을 지원합니다. 타입 파라미터는 각 호출 지점에서 추론됩니다 — `두번적용<int>(...)` 같은 명시적 호출 지점 타입 인자는 Topaz v5.2에 없습니다. 주석 달린 바인딩·파라미터·반환 위치로 타입을 고정하세요 (제네릭 타입 별칭은 데이터 타입 페이지 참고).

```topaz
function 두번적용<T>(값: T, 변환: (T) -> T) -> T {
    변환(변환(값))
}

let 다섯 = 두번적용(3, x => x + 1)             // T는 int로 추론
let 외침 = 두번적용("topaz", x => "{x}!")      // T는 string으로 추론

print("{다섯}")   // 5
print(외침)       // "topaz!!"
```

제네릭 경계, 제약, 인터페이스는 계속 deferred 상태입니다.

##  클로저

### 기본 클로저

```topaz
function 카운터만들기(초기값: int) -> () -> int {
    let mut 카운트 = 초기값  // 클로저에 의해 캡처됨
    
    return () => {
        카운트 += 1      // 외부 변수에 접근
        카운트
    }
}

let 카운터1 = 카운터만들기(0)
let 카운터2 = 카운터만들기(100)

print("{카운터1()}")  // 1
print("{카운터1()}")  // 2
print("{카운터2()}")  // 101
print("{카운터1()}")  // 3
```

### 복잡한 클로저 패턴

```topaz
// 상태를 가진 클로저
function 계좌만들기(초기잔액: float) -> { 입금: (float) -> float, 출금: (float) -> float, 잔액확인: () -> float } {
    let mut 잔액 = 초기잔액
    
    return {
        입금: (금액: float) => {
            잔액 += 금액
            잔액
        },
        출금: (금액: float) => {
            if 금액 <= 잔액 {
                잔액 -= 금액
            }
            잔액
        },
        잔액확인: () => 잔액
    }
}

let 내계좌 = 계좌만들기(1000.0)
print("{내계좌.입금(500.0)}")    // 1500.0
print("{내계좌.출금(200.0)}")    // 1300.0
print("{내계좌.잔액확인()}")      // 1300.0

// 환경 캡처하는 클로저
type 앱설정정보 = { 테마: string, 언어: string, 알림: bool }

function 설정관리자(기본설정: 앱설정정보) -> { 테마설정: (string) -> string, 테마: () -> string } {
    let mut 현재설정 = 기본설정
    
    return {
        테마설정: (테마: string) => {
            현재설정 = 현재설정{ 테마: 테마 }
            현재설정.테마
        },
        테마: () => 현재설정.테마
    }
}

let 앱설정 = 설정관리자({
    테마: "다크",
    언어: "한국어",
    알림: true
})

print(앱설정.테마())          // "다크"
앱설정.테마설정("라이트")
print(앱설정.테마())          // "라이트"
```

##  재귀 함수

### 기본 재귀

```topaz
// 팩토리얼 계산
function 팩토리얼(n: int) -> int {
    if n <= 1 {
        return 1
    }
    return n * 팩토리얼(n - 1)
}

print("{팩토리얼(5)}")  // 120

// 피보나치 수열
function 피보나치(n: int) -> int {
    if n <= 1 {
        return n
    }
    return 피보나치(n - 1) + 피보나치(n - 2)
}

print("{피보나치(8)}")  // 21

// 배열 합계 (인덱스 기반 재귀 — §22에는 슬라이싱이 없습니다)
function 부분합(배열: Array<int>, 시작: int) -> int {
    return match 배열.get(시작) {
        case Some(값) => 값 + 부분합(배열, 시작 + 1)
        case None => 0
    }
}

function 배열합계(배열: Array<int>) -> int {
    return 부분합(배열, 0)
}

print("{배열합계([1, 2, 3, 4, 5])}")  // 15
```

### 꼬리 재귀 최적화

```topaz
// 꼬리 재귀 팩토리얼
function 꼬리팩토리얼(n: int, 누적: int = 1) -> int {
    if n <= 1 {
        return 누적
    }
    return 꼬리팩토리얼(n - 1, 누적 * n)  // 꼬리 호출
}

// 꼬리 재귀 피보나치
function 꼬리피보나치(n: int, a: int = 0, b: int = 1) -> int {
    if n == 0 {
        return a
    }
    return 꼬리피보나치(n - 1, b, a + b)  // 꼬리 호출
}

print("{꼬리팩토리얼(10)}")  // 3628800
print("{꼬리피보나치(10)}")  // 55
```

##  함수형 프로그래밍 패턴

### 커링 (Currying)

```topaz
// 커링된 함수
function 더하기(a: int) -> (int) -> int {
    return (b: int) => a + b
}

let 더하기5 = 더하기(5)
print("{더하기5(3)}")  // 8

// 다중 매개변수 커링
function 곱하고더하기(곱수: float) -> (float) -> (float) -> float {
    return (더할수: float) => (값: float) => 값 * 곱수 + 더할수
}

let 두배하고십더하기 = 곱하고더하기(2.0)(10.0)
print("{두배하고십더하기(5.0)}")  // 20.0 (5 * 2 + 10)
```

### 함수 합성

```topaz
let 제곱 = (x: int) => x * x
let 두배 = (x: int) => x * 2
let 정수라벨 = (x: int) => "값: {x}"

// 값을 단계별로 파이프해서 합성
let 제곱후두배 = (x: int) => x |> 제곱() |> 두배()
let 제곱후두배후라벨 = (x: int) => x |> 제곱() |> 두배() |> 정수라벨()

print("{제곱후두배(3)}")     // 18 (3² * 2)
print(제곱후두배후라벨(4))  // "값: 32" (4² * 2 → 라벨)
```

Topaz의 공개 합성 관용구는 위처럼 `|>` 파이프라인입니다. 전용 합성 연산자는 SPEC 레벨에만 존재하며 현재 어떤 profile 표면에도 속하지 않습니다 — 연산자 레퍼런스를 참고하세요.

### 메모이제이션

```topaz
type 메모항목 = { 입력: int, 출력: int }

function 캐시찾기(캐시: Array<메모항목>, 입력: int) -> Option<int> {
    match 캐시 {
        case [] => None
        case [첫항목, ..나머지] => {
            if 첫항목.입력 == 입력 {
                Some(첫항목.출력)
            } else {
                캐시찾기(나머지, 입력)
            }
        }
    }
}

// int -> int 함수를 위한 메모이제이션 데코레이터
function 정수메모이제이션(함수: (int) -> int) -> (int) -> int {
    let mut 캐시: Array<메모항목> = []

    return (입력: int) => {
        match 캐시찾기(캐시, 입력) {
            case Some(출력) => 출력
            case None => {
                let 결과 = 함수(입력)
                캐시.push({ 입력: 입력, 출력: 결과 })
                결과
            }
        }
    }
}

function 느린피보나치(n: int) -> int {
    if n <= 1 { n } else { 느린피보나치(n - 1) + 느린피보나치(n - 2) }
}

// 느린 피보나치를 메모이제이션으로 최적화
let 메모피보나치 = 정수메모이제이션(느린피보나치)

// 이제 큰 숫자도 빠르게 계산
print("{메모피보나치(20)}")  // 빠르게 계산됨
```

> Topaz는 랭크-1 제네릭 함수 선언을 지원하지만(위 "제네릭 함수" 참고), `정수메모이제이션`은 의도적으로 구체 타입을 유지합니다: 제네릭 `memoize<T, R>`은 캐시 조회를 위해 `T`의 동등성 비교가 필요한데, 제네릭 경계·제약은 계속 deferred 상태이기 때문입니다. 같은 패턴은 다른 구체 타입에도 별도 함수로 적용할 수 있습니다.

##  함수 최적화와 베스트 프랙티스

### 1. 순수 함수 사용

```topaz
// 좋은 예: 순수 함수
function 세금계산(소득: float, 세율: float) -> float {
    return 소득 * 세율
}

// 피해야 할: 부작용이 있는 함수
let mut 전역카운터 = 0
function 나쁜함수(값: int) -> int {
    전역카운터 += 1  // 부작용!
    return 값 * 2
}

// 좋은 대안: 상태 반환
function 좋은함수(값: int, 현재카운터: int) -> { 결과: int, 새카운터: int } {
    return {
        결과: 값 * 2,
        새카운터: 현재카운터 + 1
    }
}
```

```topaz
// 명확한 함수 이름
function 사용자이메일유효성검사(이메일: string) -> bool {
    return 이메일.contains("@") && 이메일.contains(".")
}

function 주문총액계산(주문항목들: Array<{ 가격센트: int, 수량: int }>) -> int {
    return 주문항목들
        |> map(_, 항목 => 항목.가격센트 * 항목.수량)
        |> reduce(_, 0, (합계, 금액) => 합계 + 금액)
}

// 단일 책임 원칙
function 사용자데이터파싱(원시데이터: string) -> { 이름: string, 나이: int } {
    // 파싱만 담당 — toInt는 Option<int>를 반환합니다
    let 부분들 = 원시데이터.split(",")
    let 나이 = match toInt(부분들[1].trim()) {
        case Some(값) => 값
        case None => 0
    }
    return {
        이름: 부분들[0].trim(),
        나이: 나이
    }
}

function 사용자데이터검증(사용자: { 이름: string, 나이: int }) -> bool {
    // 검증만 담당
    return 사용자.이름 != "" && 사용자.나이 > 0
}
```

### 3. 에러 처리

```topaz
// Result 타입을 사용한 안전한 함수
function 안전한나누기(분자: float, 분모: float) -> Result<float, string> {
    if 분모 == 0.0 {
        return Err("0으로 나눌 수 없습니다")
    }
    return Ok(분자 / 분모)
}

// 사용 시
match 안전한나누기(10.0, 2.0) {
    case Ok(결과) => print("결과: {결과}")
    case Err(오류) => print("오류: {오류}")
}
```
